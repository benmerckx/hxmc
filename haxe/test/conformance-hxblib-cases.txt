# HXB library input cases (--hxb-lib) for haxe/test/Conformance.hx; run with:
#   bin/hxmc/hxmc -cp haxe/test --run Conformance --cases haxe/test/conformance-hxblib-cases.txt --hxb-lib
#   hxblib <name> <dir> "<library arguments>" "<arguments>"
# The reference compiles the library arguments to an HXB archive (std modules
# included); both compilers then compile the arguments against it with
# --hxb-lib, so every module the archive holds (std included) is read from it.

hxblib hxblib-basic haxe/test/hxblib/basic "-cp lib --main LibMain --interp" "-cp app --main Main --interp"
hxblib hxblib-generics haxe/test/hxblib/generics "-cp lib --main GenMain --interp" "-cp app --main Main --interp"
hxblib hxblib-macros haxe/test/hxblib/macros "-cp lib --main MacMain --interp" "-cp app --main Main --interp"
hxblib hxblib-std haxe/test/hxblib/std "-cp lib --main StdMain --interp" "-cp app --main Main --interp"
hxblib hxblib-std-dce-full haxe/test/hxblib/std "-cp lib --main StdMain --interp -dce full" "-cp app --main Main --interp -dce full"
# The library is used without its source: the application only names
# library modules.
hxblib hxblib-lib-only haxe/test/hxblib/generics "-cp lib --main GenMain --interp" "--main GenMain --interp"
# haxe/tests/misc/hl/projects/Issue12357 (compile-hxb.hxml, compile-with-hxblib.hxml), typed for HashLink without generation
hxblib hl-Issue12357 haxe/upstream/tests/misc/hl/projects/Issue12357 "-cp src -main Main -hl bin/dummy.hl" "-cp src -main Main MyException -hl bin/dummy.hl"
# Overloaded call operators (@:op(a())) of the library's abstracts, called by
# the application: the reference cannot read its own library back (a
# divergence, docs/porting.md); the port reads it.
hxblib hxblib-callop haxe/test/hxblib/callop "-cp lib --main LibMain --interp" "-cp app --main Main --interp"
diverges hxblib-callop the reader of the call operator overloads of an abstract (compiler.hxb.HxbReader.readClassFields)
